```python
RISK_MODEL_VERSION = "v1"          # module-level; stamped on every scored row (RISK-06)
KEV_FLOOR_SCORE = 90               # RISK-03 — a floor via max(), NOT an additive bonus
# 100-point budget (RESEARCH A1, tunable): severity/CVSS 35, EPSS 20, native 15, exposure 20, corroboration 10
WEIGHT_SEVERITY_CVSS = 35
WEIGHT_EPSS = 20
WEIGHT_NATIVE = 15
WEIGHT_EXPOSURE = 20
WEIGHT_CORROBORATION = 10
SEVERITY_FALLBACK = {"CRITICAL": 35, "HIGH": 20, "MEDIUM": 8, "LOW": 2, "INFO": 0}

@dataclass(frozen=True)
class FindingScoreInputs:
    """FULL input spine — every field defined here in the tracer so Plan 02 fills logic, not shape."""
    severity: str
    cvss_v3_score: Decimal | None
    epss_score: Decimal | None
    cisa_kev: bool
    source: str
    native_priority_score: Decimal | None
    native_priority_rating: str | None
    sources_count: int                 # default 1 when no VulnerabilityCorrelation row exists
    business_criticality: str
    data_sensitivity: str
    internet_facing: bool

@dataclass(frozen=True)
class RiskBreakdownComponent:
    key: str          # "severity_cvss"|"epss"|"native_exploitability"|"exposure_*"|"corroboration"
    label: str
    raw_value: str    # e.g. "0.42" | "not provided by DEFENDER"
    points: float
    max_points: float

@dataclass(frozen=True)
class RiskBreakdown:
    final_score: int
    subtotal: float
    kev_floor_applied: bool
    risk_model_version: str
    components: list[RiskBreakdownComponent]

def score_finding(inputs: FindingScoreInputs) -> RiskBreakdown:
    """Pure, no DB, deterministic. TRACER SCOPE: severity/CVSS + EPSS components are REAL;
    KEV floor is REAL. native_exploitability / exposure_* / corroboration components emit a
    row with points=0.0 and raw_value flagged '# PLAN 33-02' — Plan 02 replaces those zeros
    with real logic. final_score = max(round(subtotal), KEV_FLOOR_SCORE) if cisa_kev else round(subtotal)."""

async def compute_finding_risk_scores(db: AsyncSession, tenant_id: uuid.UUID) -> dict:
    """Full-tenant recompute (mirror compute_risk_scores shape, risk_score.py:84-147). Bulk-fetch
    all OPEN/IN_PROGRESS Vulnerability rows joined to Asset (exposure fields). TRACER: sources_count
    defaults to 1 (Plan 02 adds the VulnerabilityCorrelation bulk-join). Per row call score_finding,
    persist risk_exposure_score + risk_exposure_breakdown (serialized) + risk_model_version. Every
    query filters tenant_id. Returns {'findings_updated': int}. Does NOT roll up Asset (Plan 03)."""
```
