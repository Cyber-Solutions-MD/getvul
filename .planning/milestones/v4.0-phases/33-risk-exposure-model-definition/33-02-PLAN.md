```python
# severity/CVSS 35 (already real in tracer): cvss_v3_score is not None -> (float(cvss)/10)*35
#                                            else SEVERITY_FALLBACK[severity]
# EPSS 20 (already real in tracer): epss_score is not None -> float(epss)*20 else 0
# native-exploitability 15 (NEW): _normalize_native_signal(...) * 15
# exposure 20 (NEW): criticality*10 + internet_facing*6 + data_sensitivity*4
# corroboration 10 (NEW): min((sources_count - 1) / 3, 1.0) * 10
# KEV floor (already real): final = max(round(subtotal), 90) if cisa_kev else round(subtotal)

def _normalize_native_signal(source: str, score: Decimal | None, rating: str | None) -> float:
    """Returns 0.0-1.0. NEVER raises — soft-nulls to 0.0 on any parse failure or missing
    signal (CONTEXT native-normalization lock; mirror connectors' try/except (TypeError,ValueError))."""
    if source == "NESSUS" and score is not None:   return min(max(float(score) / 10.0, 0.0), 1.0)
    if source == "QUALYS" and score is not None:   return min(max(float(score) / 100.0, 0.0), 1.0)
    if source == "RAPID7" and score is not None:   return min(max(float(score) / 1000.0, 0.0), 1.0)
    if source == "CROWDSTRIKE" and rating is not None:
        return {"UNKNOWN":0.0,"LOW":0.25,"MEDIUM":0.5,"HIGH":0.75,"CRITICAL":1.0}.get(rating.upper(), 0.0)
    return 0.0   # DEFENDER, WIZ, missing, or unparseable — neutral, never penalized

_CRITICALITY_FRACTION = {"CRITICAL":1.0,"HIGH":0.67,"MEDIUM":0.33,"LOW":0.0}    # x10
_SENSITIVITY_FRACTION = {"RESTRICTED":1.0,"CONFIDENTIAL":0.67,"INTERNAL":0.33,"PUBLIC":0.0}  # x4
# internet_facing bool -> 1.0/0.0 x6
```
