```python
async def enqueue_backfill_job(db, tenant_id) -> RiskExposureBackfillJob:
    """Create (or return existing) the tenant's single job row, status='pending',
    rows_total_estimate = count of OPEN/IN_PROGRESS findings not yet on RISK_MODEL_VERSION.

async def process_backfill_chunk(db, tenant_id, *, now=None) -> dict:
    """Process ONE chunk for one tenant, all in ONE transaction (Pitfall 5 option (a) —
    counter advances only on commit, so a crash rolls back the whole chunk cleanly):
      1. Claim: UPDATE risk_exposure_backfill_jobs SET status='in_progress', last_heartbeat_at=now
         WHERE tenant_id=:t AND status IN ('pending','in_progress')
           AND (last_heartbeat_at IS NULL OR last_heartbeat_at < now - interval '5 min')
         RETURNING cursor_vuln_id, chunk_size.  (row is None -> already claimed/done this tick -> return)
      2. Keyset select next chunk: SELECT Vulnerability, Asset (outerjoin) WHERE tenant_id=:t
         AND status IN ('OPEN','IN_PROGRESS')
         AND (risk_model_version IS NULL OR risk_model_version != RISK_MODEL_VERSION)
         AND (cursor_vuln_id IS NULL OR Vulnerability.id > cursor_vuln_id)
         ORDER BY Vulnerability.id LIMIT chunk_size.   (empty -> mark completed + completed_at, return)
      3. Score each row via score_finding (reuse the exact FindingScoreInputs construction from
         risk_exposure_service.compute_finding_risk_scores:355-374, incl. corr_by_key bulk prefetch).
      4. ONE bulk UPDATE...FROM (VALUES ...) statement (parameterized) persisting
         risk_exposure_score + risk_exposure_breakdown + risk_model_version for the whole chunk;
         its WHERE also re-asserts v.tenant_id = :t (defense-in-depth, see threat model).
      5. Advance: cursor_vuln_id = last row id, rows_migrated += len(chunk), last_heartbeat_at=now.
      6. db.commit().  Returns {'tenant_id', 'processed': int, 'done': bool}."""

async def dispatch_backfill_chunks(db) -> dict:
    """Select tenant_ids with a job in ('pending','in_progress') UP FRONT (ids, not ORM refs);
    for each, call process_backfill_chunk inside its OWN try/except; on failure db.rollback()
    then re-select+set that tenant's job status='failed', error_message=str(e) (isolated —
    mirror poll_pending_batches:373-395 re-select-by-id-after-rollback). Returns {'tenants_advanced': n}."""
```
