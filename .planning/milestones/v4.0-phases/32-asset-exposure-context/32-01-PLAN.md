```python
EXPOSURE_FIELDS = ("business_criticality", "data_sensitivity", "internet_facing")

def infer_exposure_context(
    *, tags: list[str] | None, department: str | None,
    job_title: str | None, external_ip: str | None,
) -> tuple[str, bool, bool]:
    """Pure, no DB. Returns (business_criticality, data_sensitivity, internet_facing).
    TRACER SCOPE: business_criticality logic is REAL; data_sensitivity returns default
    'INTERNAL' and internet_facing returns default False — Plan 02 replaces those two
    default-returns with real logic. Seeds internet_facing from the 'internet-facing' tag
    is Plan 02; here it is a plain default. NEVER mutates tags."""

def apply_inference_to_asset(asset) -> list[dict]:
    """Writes each inferred field onto `asset` ONLY when asset.<field>_source == 'AUTO'
    AND the value actually changes. Returns a list of {field, old, new} change records
    (empty if nothing changed) for the caller to audit. This AUTO-gate is what makes an
    ASSET_OVERRIDE permanent across every future re-run (EXPO-03)."""

async def recompute_exposure_context(db, tenant_id) -> dict:
    """Full-tenant recompute (mirror risk_score.py::compute_risk_scores). Per asset per
    field: source == ASSET_OVERRIDE -> skip. Else auto tier via apply_inference_to_asset.
    (Plan 03 inserts the GROUP_OVERRIDE middle tier here.) Audits actual changes with
    actor system:exposure-inference. Returns {'assets_updated': int}."""
```
