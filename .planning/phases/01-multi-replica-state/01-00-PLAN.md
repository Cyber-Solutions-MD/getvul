from __future__ import annotations

import os
from typing import AsyncIterator

import pytest_asyncio
import redis.asyncio as redis
from asgi_lifespan import LifespanManager
from httpx import ASGITransport, AsyncClient


REDIS_TEST_URL = "redis://localhost:6379/1"


@pytest_asyncio.fixture(scope="function")
async def redis_test_url(monkeypatch) -> AsyncIterator[str]:
    monkeypatch.setenv("REDIS_URL", REDIS_TEST_URL)
    # Force pydantic-settings re-read inside create_app() callers.
    from app import config as _config_module
    monkeypatch.setattr(_config_module, "settings", _config_module.Settings())
    yield REDIS_TEST_URL


@pytest_asyncio.fixture(scope="function")
async def flushed_redis(redis_test_url) -> AsyncIterator[redis.Redis]:
    client = redis.from_url(redis_test_url, decode_responses=True)
    assert client.connection_pool.connection_kwargs.get("db") == 1, "tests must use db=1"
    await client.flushdb()
    try:
        yield client
    finally:
        try:
            await client.flushdb()
        except redis.RedisError:
            pass
        await client.aclose()


@pytest_asyncio.fixture(scope="function")
async def app_factory(redis_test_url):
    """Returns a callable that builds a fresh FastAPI instance bound to db=1."""
    from app.main import create_app

    def _factory():
        return create_app()

    return _factory

async def single_app(flushed_redis, app_factory):

    Yields (client, app). Used by single-replica unit tests in test_oidc_state.py

    app = app_factory()

    app_b = app_factory()

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Process startup → Redis | App lifespan opens a TCP connection on first request; bad `REDIS_URL` would fail-loud at startup. |
| Test runner → real Redis db=1 | Test fixture FLUSHDBs db=1 every run; must NEVER touch db=0 (dev/CI app db). |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-W0-flushdb-prod | T (Tampering) / DoS | `flushed_redis` fixture | mitigate | `assert connection_kwargs["db"] == 1` before `flushdb()`; URL is hard-coded `redis://localhost:6379/1`, never derived from `os.environ`. |
| T-W0-stale-settings | T (Tampering) | `redis_test_url` fixture | mitigate | `monkeypatch.setattr(config, "settings", config.Settings())` forces fresh env read; otherwise the cached singleton would still point to db=0. |
| T-W0-leaked-conn | I (Information Disclosure) / availability | `flushed_redis` fixture | mitigate | `try/finally` + `await client.aclose()` guarantees socket is returned to pool even if test panics. |
| T-W0-deprecated-close | I (DeprecationWarning surfaced as error under strict pytest) | `lifespan` shutdown | mitigate | Use `aclose()` not `close()` per redis-py 5.0.1 deprecation (RESEARCH Pitfall 5). |
| T-W0-defensive-creep | risk to maintainability | `redis_client.get_redis` | accept | No try/except — project rule. AttributeError on missing `app.state.redis` is the desired loud failure mode if Task 2 regresses. |

Block severity: high. None of the above are open at plan completion.
