```python
@router.get("/login/{provider}", response_model=AuthorizationURL)
async def login(
    provider: str,
    redis_client: Annotated[redis.Redis, Depends(get_redis)],
):
    """Initiate SSO login — returns the authorization URL to redirect the user to."""
    if provider not in ("google", "azure"):
        raise HTTPException(status_code=400, detail="Unsupported provider. Use 'google' or 'azure'.")

    state = secrets.token_urlsafe(32)
    try:
        ok = await redis_client.set(f"oidc:state:{state}", provider, ex=600, nx=True)
    except RedisError as e:
        logger.warning("redis_unavailable", subsystem="oidc_state", error=str(e))
        raise HTTPException(status_code=503, detail="Auth backend unavailable")
    if not ok:
        raise HTTPException(status_code=500, detail="State collision — retry")

    oidc = get_provider(provider)
    authorization_url = oidc.get_authorization_url(state=state)

    return AuthorizationURL(authorization_url=authorization_url, state=state)
```

Notes:
- `ex=600` is the 10-minute TTL (D-08). DO NOT use `SET` followed by `EXPIRE` — race window where the key is non-expiring (RESEARCH "Common Pitfalls — TTL drift").
- `nx=True` is defense-in-depth against the negligible 256-bit token collision (`secrets.token_urlsafe(32)`).
- `RedisError` is the parent class of `ConnectionError` and `TimeoutError` — single catch surface.

```python
@pytest.mark.asyncio
async def test_state_set_with_ttl(single_app, flushed_redis):
    client, app = single_app
    resp = await client.get("/auth/login/google")
    assert resp.status_code == 200, resp.text
    body = resp.json()
    state = body["state"]

@pytest.mark.asyncio
async def test_state_replay_rejected(single_app, flushed_redis):
    client, app = single_app
    state = "fixed-state-token-for-replay-test"
    await flushed_redis.set(f"oidc:state:{state}", "google", ex=600)

    # First callback consumes the state. It will still 4xx (token exchange against

    assert await flushed_redis.exists(f"oidc:state:{state}") == 0

    second = await client.get(f"/auth/callback/google?state={state}&code=fake")
    assert second.status_code == 400
    assert second.json()["detail"] == "Invalid or expired state parameter"


@pytest.mark.asyncio
async def test_state_provider_mismatch(single_app, flushed_redis):
    client, app = single_app
    state = "mismatch-state-token"
    await flushed_redis.set(f"oidc:state:{state}", "google", ex=600)

    resp = await client.get(f"/auth/callback/azure?state={state}&code=fake")
    assert resp.status_code == 400
    assert resp.json()["detail"] == "Invalid or expired state parameter"

    # Important: GETDEL was called even though provider mismatched, so the key
    # is deleted — preventing replay against the correct provider.
    assert await flushed_redis.exists(f"oidc:state:{state}") == 0


@pytest.mark.asyncio

async def test_login_503_on_redis_down(single_app, monkeypatch):
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → /auth/login/{provider} | Untrusted client requests an OIDC flow start; server mints a 256-bit state token. |
| IdP redirect → /auth/callback/{provider} | Untrusted code+state pair returns from external IdP; state must validate before any token exchange. |
| Backend → Redis | Internal trust; outages are availability events, not authn boundary. |

## STRIDE Threat Register

| Threat ID | Category | Component | Severity | Disposition | Mitigation |
|-----------|----------|-----------|----------|-------------|------------|
| T-OIDC-replay | S (Spoofing) — CSRF token replay | /auth/callback | high | mitigate | Atomic `GETDEL oidc:state:{state}` consumes the token in one round trip; second use returns `None` (RESEARCH §"Pattern 5" + Pitfall 8). Verified by `test_state_replay_rejected`. |
| T-OIDC-mismatch | S (Spoofing) — provider confusion | /auth/callback | high | mitigate | Compound check `stored_provider is None or stored_provider != provider` (preserved verbatim from current code per RESEARCH Pitfall 8). Even on mismatch, GETDEL has already deleted the key — preventing later replay against the correct provider. Verified by `test_state_provider_mismatch`. |
| T-OIDC-ttl | T (Tampering) — TTL drift / lingering tokens | /auth/login | medium | mitigate | `SET ... ex=600 nx=True` is one atomic command — TTL is set with the value, never via separate `SET` + `EXPIRE` (which has a non-expiring race window). Verified by `test_state_set_with_ttl` asserting TTL ∈ [595, 600]. |
| T-OIDC-redis-down | S (Spoofing via bypass) — fail-open would skip state validation | /auth/login + /auth/callback | high | mitigate | `except RedisError` returns HTTP 503; the route NEVER runs token exchange when Redis is unreachable. CSRF defense is fail-CLOSED per D-06. Verified by `test_login_503_on_redis_down`. |
| T-OIDC-info-leak | I (Information Disclosure) — distinguishing "expired" vs "tampered" | /auth/callback error message | low | mitigate | Single error string `"Invalid or expired state parameter"` regardless of branch (Pitfall 8). The compound `if stored_provider is None or stored_provider != provider` collapses both into one message. |
| T-OIDC-token-forge | S (Spoofing) — guessing a state value | /auth/login | low | accept | `secrets.token_urlsafe(32)` = 256 bits of entropy; collision/brute-force is infeasible. `nx=True` is defense-in-depth (returns None on collision → 500). Inherited from current code; not regressed. |
