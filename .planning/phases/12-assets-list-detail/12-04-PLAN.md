       'use client';
       /**
        * ChipBar — generic descriptor-driven chip filter primitive (UX-04-01 + UX-03-01).
        *
        * Phase 11 originally hardcoded severity/source axes; Phase 12 refactors to a descriptor
        * model so Assets (Category / Risk band / Source / OS) can reuse the same component.
        *
        * Behaviors preserved 1:1 from Phase 11 vuln chip-bar:
        *  - 250ms search debounce (D-F-01)
        *  - Multi-value URL state via useUrlStateList (D-F-05)
        *  - Pitfall 10 — chip click during pending debounce flushes both keys in one URL batch
        *  - Saved-filter pill is read-only (D-F-04)
        *  - Clear-all wipes all axis keys + search atomically
        *
        * T-12-05 mitigation: each ChipAxis MUST include a hardcoded allowList passed
        * straight to useUrlStateList, which clamps on BOTH read and write.
        */
       import { useCallback, useEffect, useRef, useState } from 'react';
       import { usePathname, useRouter, useSearchParams } from 'next/navigation';
       import { useUrlStateList } from '@/hooks/use-url-state-list';
       import { cn } from '@/lib/utils';

       export type ChipDescriptor = {
         value: string;
         label: string;
         glyph?: string;
         glyphClassName?: string;
       };

       export type ChipAxis = {
         key: string;
         label?: string;
         allowList: readonly string[];
         chips?: ChipDescriptor[];
         counts?: Record<string, number>;
         derivedFromCounts?: boolean;
       };

       export type ChipBarProps = {
         axes: ChipAxis[];
         savedFilter?: { label: string; query: string } | null;
         showSearch?: boolean;
         searchPlaceholder?: string;
       };

       const SEARCH_DEBOUNCE_MS = 250;

       function ChipGroup({
         axis,
         onChipFlush,
       }: {
         axis: ChipAxis;
         onChipFlush: () => void;
       }) {
         const [value, , toggle] = useUrlStateList<string>(axis.key, axis.allowList, []);

         // Decide chip set: explicit chips prop, or derived from counts keys.
         const chips: ChipDescriptor[] = axis.derivedFromCounts
           ? Object.keys(axis.counts ?? {})
               .filter((k) => (axis.allowList as readonly string[]).includes(k))
               .map((k) => ({ value: k, label: k }))
           : (axis.chips ?? []);

         return (
           <div className="flex flex-wrap items-center gap-1.5" data-axis={axis.key}>
             {axis.label && (
               <span className="mr-1 text-xs uppercase tracking-wide text-text-subtle">{axis.label}</span>
             )}
             {chips.map((c) => {
               const active = value.includes(c.value);
               const count = axis.counts?.[c.value];
               const display = typeof count === 'number' ? `${c.label} · ${count}` : c.label;
               return (
                 <button
                   key={c.value}
                   type="button"
                   onClick={() => {
                     toggle(c.value);
                     onChipFlush();
                   }}
                   data-active={active}
                   className={cn(
                     'inline-flex items-center gap-1.5 rounded-full border px-2.5 py-1 text-xs',
                     active
                       ? 'border-border bg-surface-2 text-text'
                       : 'border-border-subtle bg-transparent text-text-subtle hover:text-text',
                   )}
                 >
                   {c.glyph && <span className={cn('font-mono', c.glyphClassName)}>{c.glyph}</span>}
                   <span>{display}</span>
                 </button>
               );
             })}
           </div>
         );
       }

       export function ChipBar({
         axes,
         savedFilter,
         showSearch = true,
         searchPlaceholder = 'Search…',
       }: ChipBarProps) {
         const router = useRouter();
         const pathname = usePathname();
         const params = useSearchParams();

         const initialSearch = params?.get('search') ?? '';
         const [searchInput, setSearchInput] = useState<string>(initialSearch);
         const lastFlushedRef = useRef<string>(initialSearch);

         useEffect(() => {
           const urlSearch = params?.get('search') ?? '';
           if (urlSearch === '' && lastFlushedRef.current !== '') {
             setSearchInput('');
             lastFlushedRef.current = '';
           }
         }, [params]);

         const buildHref = useCallback(
           (mutate: (sp: URLSearchParams) => void): string => {
             const sp = new URLSearchParams(params?.toString() ?? '');
             mutate(sp);
             const qs = sp.toString();
             return qs ? `${pathname}?${qs}` : (pathname ?? '/');
           },
           [params, pathname],
         );

         // D-F-01 — debounced search flush.
         useEffect(() => {
           if (!showSearch) return;
           if (searchInput === lastFlushedRef.current) return;
           const t = setTimeout(() => {
             const href = buildHref((sp) => {
               if (searchInput === '') sp.delete('search');
               else sp.set('search', searchInput);
             });
             lastFlushedRef.current = searchInput;
             router.replace(href, { scroll: false });
           }, SEARCH_DEBOUNCE_MS);
           return () => clearTimeout(t);
         }, [searchInput, buildHref, router, showSearch]);

         // Pitfall 10 — chip click flushes the pending search synchronously alongside.
         const onChipFlush = useCallback(() => {
           if (searchInput === lastFlushedRef.current) return;
           const href = buildHref((sp) => {
             if (searchInput === '') sp.delete('search');
             else sp.set('search', searchInput);
           });
           lastFlushedRef.current = searchInput;
           router.replace(href, { scroll: false });
         }, [searchInput, buildHref, router]);

         const clearAll = () => {
           setSearchInput('');
           lastFlushedRef.current = '';
           const href = buildHref((sp) => {
             axes.forEach((a) => sp.delete(a.key));
             sp.delete('search');
           });
           router.replace(href, { scroll: false });
         };

         const applySavedFilter = () => {
           if (!savedFilter) return;
           const sp = new URLSearchParams(savedFilter.query);
           const href = buildHref((target) => {
             axes.forEach((a) => target.delete(a.key));
             target.delete('search');
             sp.forEach((v, k) => target.append(k, v));
           });
           router.replace(href, { scroll: false });
         };

         return (
           <div role="search" className="flex flex-wrap items-center gap-3" data-chip-bar="generic">
             {showSearch && (
               <input
                 type="search"
                 value={searchInput}
                 onChange={(e) => setSearchInput(e.target.value)}
                 placeholder={searchPlaceholder}
                 className="w-64 rounded-md border border-border-subtle bg-surface-2 px-3 py-1.5 text-sm text-text placeholder:text-text-subtle focus-visible:outline focus-visible:outline-2 focus-visible:outline-violet"
                 aria-label="Search"
               />
             )}
             {axes.map((axis) => (
               <ChipGroup key={axis.key} axis={axis} onChipFlush={onChipFlush} />
             ))}
             {savedFilter && (
               <button
                 type="button"
                 onClick={applySavedFilter}
                 className="rounded-full border border-violet/40 bg-violet/10 px-2.5 py-1 text-xs text-violet"
                 data-saved-filter-pill
               >
                 ★ {savedFilter.label}
               </button>
             )}
             <button
               type="button"
               onClick={clearAll}
               className="ml-auto text-xs text-text-subtle hover:text-text"
             >
               Clear all
             </button>
           </div>
         );
       }
       ```

       'use client';
       /**
        * VulnerabilitiesChipBar — Phase 12 thin wrapper over the generic <ChipBar>.
        *
        * Preserves the exported name `ChipBar` and the `{ facets }` prop so the existing
        * /vulnerabilities page callsite is unchanged. Internally constructs the vuln-axes
        * descriptor and forwards to components/ui/ChipBar.
        */
       import { ChipBar as GenericChipBar, type ChipAxis } from '@/components/ui/ChipBar';
       import { microcopy } from './microcopy';
       import { useSavedFilters } from '@/lib/queries/use-saved-filters';

       const SEVERITIES = ['critical', 'high', 'medium', 'low', 'info'] as const;
       const SOURCES = ['QUALYS', 'TENABLE', 'RAPID7', 'CROWDSTRIKE', 'AWS_INSPECTOR', 'WIZ', 'MOCK'] as const;

       export type ChipBarFacets = {
         severity: Record<string, number>;
         source: Record<string, number>;
         status?: Record<string, number>;
       };

       type Props = { facets: ChipBarFacets };

       const SEV_GLYPH: Record<(typeof SEVERITIES)[number], string> = {
         critical: '■', high: '▲', medium: '◆', low: '○', info: '□',
       };

       const SEV_TINT: Record<(typeof SEVERITIES)[number], string> = {
         critical: 'text-severity-critical',
         high: 'text-severity-high',
         medium: 'text-severity-medium',
         low: 'text-severity-low',
         info: 'text-severity-info',
       };

       export function ChipBar({ facets }: Props) {
         const savedFilters = useSavedFilters();
         const firstSaved = savedFilters.data?.[0];

         const severityAxis: ChipAxis = {
           key: 'severity',
           label: 'Severity',
           allowList: SEVERITIES,
           counts: facets.severity,
           chips: SEVERITIES.map((s) => ({
             value: s,
             label: microcopy.chips[s as keyof typeof microcopy.chips] ?? s,
             glyph: SEV_GLYPH[s],
             glyphClassName: SEV_TINT[s],
           })),
         };

         const sourceAxis: ChipAxis = {
           key: 'source',
           label: 'Source',
           allowList: SOURCES,
           counts: facets.source,
           derivedFromCounts: true,
         };

         const axes: ChipAxis[] = [severityAxis, sourceAxis];

         return (
           <GenericChipBar
             axes={axes}
             savedFilter={
               firstSaved
                 ? {
                     label: (firstSaved as { name?: string }).name ?? 'Saved',
                     query: (firstSaved as { query?: string }).query ?? '',
                   }
                 : null
             }
           />
         );
       }
       ```
