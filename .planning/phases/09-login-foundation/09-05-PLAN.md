CRITICAL CORRECTION: `window.location.href = X` does NOT throw synchronously on backend 5xx — the browser just navigates (or stays put on a 5xx response page). So the implicit catch alone never fires the D-51 path for "backend returns 5xx", which is the scenario T-09-05-07 claims to mitigate. Fix: pre-flight the SSO endpoint with `HEAD` BEFORE the `window.location.href` assignment.

```ts
/** Initiates OIDC handshake. Accepted values: 'google' | 'azure'.
 *  Microsoft SSO maps to 'azure' at the UI layer (SsoRow handler in /login). */
async function loginSSO(provider: 'google' | 'azure') {
  const provName = provider === 'google' ? 'Google' : 'Microsoft';
  const apiUrl = process.env.NEXT_PUBLIC_API_URL ?? '';
  try {
    // D-51 reachability fix: HEAD pre-flight catches backend 5xx BEFORE we navigate away.
    // Without this, window.location.href = X never throws on a 5xx response — the browser
    // just navigates to the 5xx page and the SSO-unavailable copy is unreachable.
    const probe = await fetch(`${apiUrl}/auth/login/${provider}`, { method: 'HEAD' });
    if (!probe.ok) {
      throw new Error('SSO_UNAVAILABLE');
    }
    // OK — kick off the OIDC handshake (page unloads on success).
    window.location.href = `${apiUrl}/auth/login/${provider}`;
  } catch (e: unknown) {
    // D-51 verbatim surfacing: caller (/login SsoRow) catches and shows ErrorAlert.
    throw new Error(`Sign-in with ${provName} is temporarily unavailable. Try email instead.`);
  }
}
```

NOTE: the HEAD pre-flight assumes the backend's OIDC route accepts HEAD (FastAPI's default routing accepts HEAD on GET routes via Starlette). If the backend rejects HEAD with 405, that triggers `!probe.ok` and the user sees the D-51 copy — strictly more conservative than the v1 silent-failure. If this becomes a false-positive issue post-Phase 9, swap HEAD for GET with `redirect: 'manual'` and accept any 3xx as OK (a future tightening; the current shape is unambiguously correct for the "backend returns 5xx" scenario).

```ts
import { NextResponse, type NextRequest } from 'next/server';

// Legacy URL redirects — the 5 root-level routes deleted in Wave 2 (D-34)
// resolved to /dashboard/* canonical paths. Preserve bookmarks.
const LEGACY_MAP: Record<string, string> = {
  '/assets':         '/dashboard/assets',
  '/integrations':   '/dashboard/integrations',
  '/settings':       '/dashboard/settings',
  '/tickets':        '/dashboard/tickets',
  '/vulnerabilities':'/dashboard/vulnerabilities',
};

export function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;

  // Exact root match
  if (LEGACY_MAP[pathname]) {
    const url = request.nextUrl.clone();
    url.pathname = LEGACY_MAP[pathname];
    return NextResponse.redirect(url, 308);  // permanent redirect; preserves method
  }

  // Subpath: /assets/* → /dashboard/assets/*
  for (const [legacy, canonical] of Object.entries(LEGACY_MAP)) {
    if (pathname.startsWith(legacy + '/')) {
      const url = request.nextUrl.clone();
      url.pathname = canonical + pathname.slice(legacy.length);
      return NextResponse.redirect(url, 308);
    }
  }

  return NextResponse.next();
}

export const config = {
  matcher: ['/assets/:path*', '/integrations/:path*', '/settings/:path*', '/tickets/:path*', '/vulnerabilities/:path*'],
};
```

NOTE: this middleware does NOT do auth gating — research §Pattern 5 explained why (cookie-vs-localStorage). It only handles legacy URL redirects.

```tsx
  const [mode, setMode] = useState<Mode>(resetToken ? 'reset' : 'login');
  const [authError, setAuthError] = useState<string | null>(null);
  const [forgotSent, setForgotSent] = useState(false);

  // Bounce already-authed users with ?next= validated (D-50, Pitfall 10).
  useEffect(() => {
    if (user) {
      router.replace(sanitizeNext(searchParams.get('next')));
    }
  }, [user, router, searchParams]);

  return (
    <main className="min-h-screen grid grid-cols-1 lg:grid-cols-2 bg-bg text-text">
      <LeftPanel />
      <RightPanel
        mode={mode}
        setMode={(m) => { setMode(m); setAuthError(null); setForgotSent(false); }}
        authError={authError}
        setAuthError={setAuthError}
        forgotSent={forgotSent}
        setForgotSent={setForgotSent}
        login={login}
        loginSSO={loginSSO}
        router={router}
        searchParams={searchParams}
        resetToken={resetToken}
      />
    </main>
  );
}

// D-46 UI label "Microsoft" maps to backend OIDC route name 'azure' — backend route is /auth/login/azure (existing).

const toBackendProvider = (ui: 'google' | 'microsoft'): 'google' | 'azure' =>
  ui === 'microsoft' ? 'azure' : 'google';

function SsoRow({ loginSSO, setAuthError }: RightPanelProps) {
  async function handleSso(uiProvider: 'google' | 'microsoft') {
    setAuthError(null);
    // D-46 UI label "Microsoft" maps to backend OIDC route name 'azure' — never pass 'microsoft' to loginSSO.
    const backendProvider = toBackendProvider(uiProvider);
    try {
      await loginSSO(backendProvider);
    } catch (e: unknown) {
      // D-51: loginSSO now throws with the user-facing message
      const msg = e instanceof Error ? e.message
        : `Sign-in with ${uiProvider === 'google' ? 'Google' : 'Microsoft'} is temporarily unavailable. Try email instead.`;
      setAuthError(msg);
    }
  }

  return (
    <>
      <div className="space-y-2.5">
        <SsoButton provider="google" onClick={() => handleSso('google')} />
        <SsoButton provider="microsoft" onClick={() => handleSso('microsoft')} />
      </div>
      {/* `or with email` divider per D-32, UX-01-02 — one-off, not a primitive */}
      <div className="flex items-center gap-3 text-xs uppercase tracking-wider text-text-faint">
        <span className="h-px flex-1 bg-border-subtle" />
        <span>or with email</span>
        <span className="h-px flex-1 bg-border-subtle" />
      </div>
    </>
  );
}

function LoginForm({ login, setAuthError, router, searchParams, setMode }: RightPanelProps) {
  const form = useForm<z.infer<typeof loginSchema>>({
    resolver: zodResolver(loginSchema),
    mode: 'onSubmit',  // D-53
    defaultValues: { email: '', password: '' },
  });

  async function onSubmit(values: z.infer<typeof loginSchema>) {
    setAuthError(null);
    try {
      await login(values.email, values.password);
      // Redirect to ?next= with open-redirect mitigation per D-50, Pitfall 10
      const dest = sanitizeNext(searchParams.get('next'));
      router.replace(dest);
    } catch (e: unknown) {
      // D-49: 401 → generic; other 4xx → pass-through backend message
      const err = e as { status?: number; message?: string } | undefined;
      const status = err?.status ?? (err as any)?.response?.status;
      if (status === 401) {
        setAuthError('Email or password is incorrect.');
      } else {
        setAuthError(err?.message ?? 'Sign-in failed. Try again in a moment.');
      }
    }
  }

  return (
    <Form {...form}>
      <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4">
        <FormField
          control={form.control}
          name="email"
          render={({ field }) => (
            <FormItem>
              <FormLabel>Email</FormLabel>
              <FormControl>
                <Input
                  type="email"
                  placeholder="you@company.com"
                  autoFocus
                  autoComplete="email"
                  {...field}
                />
              </FormControl>
              <FormMessage />
            </FormItem>
          )}
        />

        <FormField
          control={form.control}
          name="password"
          render={({ field }) => (
            <FormItem>
              <FormLabel>Password</FormLabel>
              <FormControl>
                <Input
                  type="password"
                  autoComplete="current-password"
                  {...field}
                />
              </FormControl>
              <FormMessage />
            </FormItem>
          )}
        />

        {/* D-47: right-aligned "Forgot password?" link below password */}
        <div className="flex justify-end">
          <button
            type="button"
            className="text-sm text-text-muted hover:text-text underline-offset-4 hover:underline"
            onClick={() => setMode('forgot')}
          >
            Forgot password?
          </button>
        </div>

        {/* D-52: per-mode CTA copy */}
        <Button
          type="submit"
          variant="cta"
          size="lg"
          className="w-full"
          loading={form.formState.isSubmitting}
          loadingText="Signing in…"
        >
          Sign in
        </Button>
      </form>
    </Form>
  );
}

function ForgotForm({ setAuthError, forgotSent, setForgotSent }: RightPanelProps) {
  const form = useForm<z.infer<typeof forgotSchema>>({
    resolver: zodResolver(forgotSchema),
    mode: 'onSubmit',
    defaultValues: { email: '' },
  });

  async function onSubmit(values: z.infer<typeof forgotSchema>) {
    setAuthError(null);
    try {
      // Direct backend call — we don't go through useAuth() for forgot (existing v1 pattern preserved).
      // Pitfall 9: regardless of backend response shape, we show the same generic copy.
      const apiUrl = process.env.NEXT_PUBLIC_API_URL ?? '';
      await fetch(`${apiUrl}/auth/forgot-password`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email: values.email }),
      });
    } catch {
      // Network failure — still show generic to avoid signal even on infra incidents
    } finally {
      setForgotSent(true);
    }
  }

  if (forgotSent) {
    // Pitfall 9: anti-enumeration — generic confirmation regardless of actual outcome
    return (
      <div className="rounded-md border border-border bg-surface p-4 text-sm text-text-muted">
        If that email is registered, a reset token is on its way.

  }

  return (
    <Form {...form}>
      <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4">
        <FormField
          control={form.control}
          name="email"
          render={({ field }) => (

              <FormLabel>Email</FormLabel>

                  autoFocus
                  autoComplete="email"
                  {...field}
                />
              </FormControl>
              <FormMessage />
            </FormItem>
          )}
        />

          type="submit"

          size="lg"
          className="w-full"

}

function ResetForm({ resetToken, setAuthError, setMode }: RightPanelProps) {
  const form = useForm<z.infer<typeof resetSchema>>({
    resolver: zodResolver(resetSchema),
    mode: 'onSubmit',
    defaultValues: { token: resetToken ?? '', newPassword: '' },
  });

  const [done, setDone] = useState(false);

  async function onSubmit(values: z.infer<typeof resetSchema>) {
    setAuthError(null);
    try {
      const apiUrl = process.env.NEXT_PUBLIC_API_URL ?? '';
      const res = await fetch(`${apiUrl}/auth/reset-password`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ token: values.token, new_password: values.newPassword }),
      });
      if (!res.ok) {
        const body = await res.json().catch(() => ({}));
        setAuthError(body?.detail ?? 'Reset failed. Request a new link.');
        return;
      }
      setDone(true);
    } catch {
      setAuthError('Network error. Try again in a moment.');
    }
  }

  if (done) {
    return (
      <div className="space-y-4">
        <div className="rounded-md border border-success bg-success-soft p-4 text-sm text-success">
          Password updated. Sign in to continue.
        </div>
        <Button variant="secondary" size="lg" className="w-full" onClick={() => setMode('login')}>
          Back to sign in
        </Button>
      </div>
    );
  }

  return (
    <Form {...form}>
      <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4">
        <FormField
          control={form.control}
          name="token"
          render={({ field }) => (
            <FormItem>
              <FormLabel>Reset token</FormLabel>
              <FormControl>
                <Input
                  type="text"
                  // D-48: token paste field uses autoComplete='off'
                  autoComplete="off"
                  spellCheck={false}
                  {...field}
                />
              </FormControl>
              <FormMessage />
            </FormItem>
          )}
        />

        <FormField
          control={form.control}
          name="newPassword"
          render={({ field }) => (
            <FormItem>
              <FormLabel>New password</FormLabel>
              <FormControl>
                <Input
                  type="password"
                  // D-48 spec says 'first field' — for reset mode the first FILLABLE field is newPassword
                  // (token is pre-filled from ?reset= deep link, so autoFocus belongs on newPassword).
                  autoFocus
                  autoComplete="new-password"
                  {...field}
                />
              </FormControl>
              <FormMessage />
            </FormItem>
          )}
        />

        <Button
          type="submit"
          variant="cta"
          size="lg"
          className="w-full"
          loading={form.formState.isSubmitting}
          loadingText="Updating…"
        >
          Set new password
        </Button>
      </form>
    </Form>
  );
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| URL `?next=` query param ↔ post-login redirect | User-controllable input; must be sanitized to same-origin before `router.replace()` |
| URL `?reset=TOKEN` ↔ form pre-fill | Token rendered into Input's `value`; React text-escaped — no XSS sink |
| Backend response shape ↔ ErrorAlert copy | Backend `detail` is rendered as text content (not HTML); D-49 enforces generic copy on 401 to prevent user enumeration |
| `loginSSO` thrown errors ↔ ErrorAlert | Caller catches and surfaces message; provider-name-dependent copy per D-51 |

| T-09-05-01 | T (Tampering) | Open redirect via `?next=//evil.com` | mitigate | `sanitizeNext()` validates same-origin per Pitfall 10: `decoded.startsWith('/') && !decoded.startsWith('//') && !decoded.startsWith('/\\\\')`. Unit-tested in page.test.tsx for protocol-relative, absolute URL, backslash trickery, decode failure cases. |
