**TrendChart (`trend-chart.tsx`)** — D-C-01..10 + Pattern 10 verbatim base, with the sr-only table, custom tooltip, range toggle, and `aria-hidden` SVG wired together:

```tsx
'use client';

import { BarChart, Bar, XAxis, YAxis, Tooltip, ResponsiveContainer, CartesianGrid } from 'recharts';
import { useMemo } from 'react';
import { cn } from '@/lib/utils';
import { microcopy } from '@/components/dashboard/microcopy';
import { usePrefersReducedMotion } from '@/hooks/use-prefers-reduced-motion';

export type TrendDatum = {
  date: string;
  critical: number; high: number; medium: number; low: number;
};
export type Range = '7d' | '30d' | '90d';
export type TrendChartProps = {
  data: TrendDatum[];
  range: Range;
  onRangeChange: (next: Range) => void;
};

const SEVERITY_FILLS = {
  critical: 'var(--color-severity-critical)',
  high:     'var(--color-severity-high)',
  medium:   'var(--color-severity-medium)',
  low:      'var(--color-severity-low)',
} as const;

const SEVERITY_GLYPHS = {
  critical: '■',
  high:     '▲',
  medium:   '◆',
  low:      '○',
} as const;

function fmtTick(iso: string): string {
  // 'Apr 16' — short month + day in browser locale
  const d = new Date(iso);
  return d.toLocaleDateString(undefined, { month: 'short', day: 'numeric' });
}

function fmtFullDate(iso: string): string {
  const d = new Date(iso);
  return d.toLocaleDateString(undefined, { weekday: 'short', month: 'short', day: 'numeric' });
}

function isLatestDay(label: string, lastDate: string): boolean {
  return label === lastDate;
}

function SeverityTooltip({ active, payload, label, lastDate }: any) {
  if (!active || !payload?.length) return null;
  const get = (k: string) => payload.find((p: any) => p.dataKey === k)?.value ?? 0;
  const total = (['critical','high','medium','low'] as const).reduce((s, k) => s + get(k), 0);
  return (
    <div role="tooltip" className="rounded-md border border-border bg-surface px-3 py-2 shadow-card">
      <p className="text-xs text-text-muted">{isLatestDay(label, lastDate) ? microcopy.trend.todaySoFar : fmtFullDate(label)}</p>
      <p className="font-mono text-sm text-text">{total} open</p>
      <ul className="mt-1 space-y-0.5 font-mono text-xs">
        <li><span className="text-severity-critical" aria-hidden="true">{SEVERITY_GLYPHS.critical}</span> Critical: {get('critical')}</li>
        <li><span className="text-severity-high"     aria-hidden="true">{SEVERITY_GLYPHS.high}</span> High: {get('high')}</li>
        <li><span className="text-severity-medium"   aria-hidden="true">{SEVERITY_GLYPHS.medium}</span> Medium: {get('medium')}</li>
        <li><span className="text-severity-low"      aria-hidden="true">{SEVERITY_GLYPHS.low}</span> Low: {get('low')}</li>
      </ul>
    </div>
  );
}

function RangeToggle({ value, onChange }: { value: Range; onChange: (n: Range) => void }) {
  const opts: { id: Range; label: string }[] = [
    { id: '7d',  label: microcopy.trend.range7d  },
    { id: '30d', label: microcopy.trend.range30d },
    { id: '90d', label: microcopy.trend.range90d },
  ];
  return (
    <div role="group" aria-label="Trend range" className="inline-flex rounded-md border border-border-subtle p-0.5">
      {opts.map(o => {
        const active = o.id === value;
        return (
          <button
            key={o.id}
            type="button"
            aria-pressed={active}
            onClick={() => onChange(o.id)}
            className={cn(
              'rounded-sm px-3 py-1 text-xs font-mono transition-colors',
              active ? 'bg-surface-2 text-text' : 'text-text-muted hover:text-text',
            )}
          >
            {o.label}
          </button>
        );
      })}
    </div>
  );
}

function ChartDataTable({ data }: { data: TrendDatum[] }) {
  return (
    <table className="sr-only" aria-label={microcopy.trend.h2}>
      <caption>Daily counts of open vulnerabilities by severity</caption>
      <thead>
        <tr>
          <th scope="col">Date</th>
          <th scope="col">Critical</th>
          <th scope="col">High</th>
          <th scope="col">Medium</th>
          <th scope="col">Low</th>
          <th scope="col">Total</th>
        </tr>
      </thead>
      <tbody>
        {data.map(d => (
          <tr key={d.date}>
            <th scope="row">{fmtFullDate(d.date)}</th>
            <td>{d.critical}</td><td>{d.high}</td><td>{d.medium}</td><td>{d.low}</td>
            <td>{d.critical + d.high + d.medium + d.low}</td>
          </tr>
        ))}
      </tbody>
    </table>
  );
}

export function TrendChart({ data, range, onRangeChange }: TrendChartProps) {
  const lastDate = data[data.length - 1]?.date ?? '';
  const reduced = usePrefersReducedMotion();

  return (
    <div className="space-y-3">
      <div className="flex items-center justify-between">
        <h3 className="sr-only">{microcopy.trend.h2}</h3>
        <RangeToggle value={range} onChange={onRangeChange} />
      </div>
      <div aria-hidden="true">
        <ResponsiveContainer width="100%" height={200}>
          <BarChart data={data} margin={{ top: 8, right: 8, left: 0, bottom: 8 }} accessibilityLayer>
            <CartesianGrid strokeDasharray="2 4" vertical={false} stroke="var(--color-border-subtle)" />
            <XAxis dataKey="date" tickFormatter={fmtTick} stroke="var(--color-text-muted)" />
            <YAxis stroke="var(--color-text-muted)" />
            <Tooltip content={<SeverityTooltip lastDate={lastDate} />} cursor={{ fill: 'var(--color-surface-2)' }} />
            {/* Same stackId → all 4 stack. Order = paint order = stack order (low at base). */}
            <Bar dataKey="low"      stackId="s" fill={SEVERITY_FILLS.low}      isAnimationActive={reduced ? false : 'auto' as any} />
            <Bar dataKey="medium"   stackId="s" fill={SEVERITY_FILLS.medium}   isAnimationActive={reduced ? false : 'auto' as any} />
            <Bar dataKey="high"     stackId="s" fill={SEVERITY_FILLS.high}     isAnimationActive={reduced ? false : 'auto' as any} />
            <Bar dataKey="critical" stackId="s" fill={SEVERITY_FILLS.critical} isAnimationActive={reduced ? false : 'auto' as any} />
          </BarChart>
        </ResponsiveContainer>
      </div>
      <ChartDataTable data={data} />
    </div>
  );
}
```

Notes the executor must respect:
- The `aria-hidden="true"` wrapper around `<ResponsiveContainer>` is intentional (D-Ax-03 — screen readers consume the table, not the SVG). The chart is a visual aid; the table is the canonical data.

- All colors are CSS variables. No hex.
- Range toggle uses `aria-pressed` for the active state — accessible toggle pattern.
- The h3 is `sr-only` because the Plan-05 wrapper renders a section-level h2.

```js
#!/usr/bin/env node
// check-bundle.mjs — D-Perf-01 enforcer
// Usage:
//   node scripts/check-bundle.mjs --route /dashboard --max-kb 180
//   next build | node scripts/check-bundle.mjs --route /dashboard --max-kb 180
// Exit codes:
//   0 — within budget
//   1 — over budget
//   2 — route not found in build output

import { execSync } from 'node:child_process';
import process from 'node:process';

function parseArgs(argv) {
  const args = {};
  for (let i = 0; i < argv.length; i++) {
    if (argv[i] === '--route') args.route = argv[++i];
    else if (argv[i] === '--max-kb') args.maxKb = Number(argv[++i]);
  }
  if (!args.route) {
    console.error('check-bundle: --route is required (e.g., --route /dashboard)');
    process.exit(2);
  }
  if (!args.maxKb) {
    console.error('check-bundle: --max-kb is required (integer kilobytes)');
    process.exit(2);
  }
  return args;
}

function readBuildOutput() {
  // If stdin is a pipe (not a TTY) and has data, read from it.
  // Otherwise, invoke `next build` and capture stdout.
  if (!process.stdin.isTTY) {
    let data = '';
    try {
      data = require('node:fs').readFileSync(0, 'utf-8');
      if (data && data.trim().length > 0) return data;
    } catch {}
  }
  try {
    return execSync('npx next build', { encoding: 'utf-8', stdio: ['ignore', 'pipe', 'inherit'] });
  } catch (e) {
    console.error('check-bundle: next build failed');
    process.exit(2);
  }
}

function parseRouteLine(output, route) {
  // Next.js 15 build output format (per Pitfall 4):
  //   Route (app)                              Size      First Load JS
  //   ┌ ○ /                                    1.2 kB    178 kB
  //   ├ ○ /dashboard                           5.4 kB    178 kB
  //   └ ○ /login                               2.1 kB    145 kB
  //   + First Load JS shared by all            87 kB
  //     ├ chunks/main-xxx.js                   45 kB
  //   ...
  //
  // We want the per-route "First Load JS" cell, NOT the "shared by all" footer.
  const lines = output.split('\n');
  // Match a line whose route token matches /dashboard exactly (boundary on whitespace),
  // and pull the LAST "<num> <unit>B" cell on that line.
  const routeRegex = new RegExp(`(?:^|[│├└─\\s])${route.replace(/\\//g, '\\/')}(?=\\s|$)`);
  for (const line of lines) {
    if (routeRegex.test(line)) {
      // Find all "<num> kB|MB|B" tokens; the last one is "First Load JS"
      const tokens = [...line.matchAll(/(\d+(?:\.\d+)?)\s*(kB|MB|B)/gi)];
      if (tokens.length >= 2) {
        const [, num, unit] = tokens[tokens.length - 1];
        const bytes = Number(num) * (unit === 'MB' ? 1_048_576 : unit === 'kB' ? 1024 : 1);
        return { line: line.trim(), bytes };
      }
    }
  }
  return null;
}

const args = parseArgs(process.argv.slice(2));
const output = readBuildOutput();
const match = parseRouteLine(output, args.route);

if (!match) {
  console.error(`check-bundle: could not find route ${args.route} in build output.`);
  process.exit(2);
}

const budgetBytes = args.maxKb * 1024;
const kb = (match.bytes / 1024).toFixed(1);
const budgetKb = args.maxKb.toFixed(1);

if (match.bytes > budgetBytes) {
  console.error(`check-bundle: FAIL — ${args.route} First-Load JS is ${kb} kB (> ${budgetKb} kB budget)`);
  console.error(`  parsed line: ${match.line}`);
  process.exit(1);
}

console.log(`check-bundle: OK — ${args.route} First-Load JS is ${kb} kB (≤ ${budgetKb} kB budget)`);
process.exit(0);
```
