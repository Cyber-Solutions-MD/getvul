**`hero.tsx`** — D-H-01..12 + microcopy + ErrorBoundary-friendly:

```tsx
'use client';
import { Zap, Clock } from 'lucide-react';
import { Button } from '@/components/ui/button';
import { microcopy } from './microcopy';
import { useSnoozeMutation } from '@/lib/mutations/use-snooze';
import { useUndoSnoozeMutation } from '@/lib/mutations/use-undo-snooze';
import { useStats } from '@/lib/queries/use-stats';
import { useToast } from '@/components/ui/ToastProvider';
import Link from 'next/link';
import { cn } from '@/lib/utils';

export function Hero() {
  const stats = useStats();
  const snooze = useSnoozeMutation();
  const undoSnooze = useUndoSnoozeMutation();
  const { toast } = useToast();  // Plan 02 API: useToast() returns { toast: (input) => void }

  if (stats.isPending) {
    return (
      <section aria-busy="true" aria-labelledby="hero-h" className="h-40 rounded-lg bg-surface-2 animate-pulse">
        <h2 id="hero-h" className="sr-only">Action required</h2>
      </section>
    );
  }
  if (stats.error) {
    // D-E-02 inline error fallback (Hero's own; sibling ErrorBoundary catches throws but this is a query error)
    const code = (stats.error as any)?.code ?? 'unknown';
    const reqId = (stats.error as any)?.requestId ?? 'unknown';
    return (
      <section className="rounded-lg border border-danger bg-danger-soft p-5 text-sm" role="alert">
        {microcopy.error.inline('Hero', code, reqId)}
      </section>
    );
  }

  const data = stats.data!;
  const n = data.dashboard_tiles.critical_open.value;
  const topVuln = data.top_vuln;

  if (n === 0) {
    // D-H-09 quiet-win — green dot, no Snooze CTA, no sub-line
    return (
      <section aria-labelledby="hero-h" className="rounded-lg border border-border-subtle bg-surface p-6">
        <div className="mb-2 flex items-center gap-2">
          <span className="block h-2 w-2 rounded-full bg-success" aria-hidden="true" />
          <span className="text-xs uppercase tracking-wide text-text-muted">Status</span>
        </div>
        <h2 id="hero-h" className="text-3xl font-semibold text-text">{microcopy.hero.quietWin}</h2>
      </section>
    );
  }

  const headline = n === 1 ? microcopy.hero.headlineSingular : microcopy.hero.headlinePlural(n);
  const subLine = topVuln
    ? microcopy.hero.subLineTemplate(topVuln.host, topVuln.path, Number(topVuln.cvss), topVuln.exploited)
    : null;

  const onSnooze = async () => {
    if (!topVuln) return;
    try {
      // Blocker 2 fix: snooze against the vuln UUID, NOT cve_id. top_vuln.id is required from Plan 01.
      await snooze.mutateAsync({ id: topVuln.id });
      // Blocker 1 fix: real Toast API per Plan 02 (`toast({...})` with duration + action slot).
      // 8s duration = D-H-08 undo window. Action slot fires the symmetric unsnooze mutation.
      toast({
        title: microcopy.snooze.toastTitle(topVuln.cve_id),
        message: microcopy.snooze.toastMessage(topVuln.cve_id),
        variant: 'success',
        duration: 8000,
        action: {
          label: microcopy.snooze.toastActionLabel,  // 'Undo'
          onClick: () => undoSnooze.mutate({ id: topVuln.id }),
        },
      });
    } catch (e: any) {
      toast({
        message: microcopy.snooze.toastError(e?.status ?? 'unknown'),
        variant: 'error',
      });
    }
  };

  return (
    <section aria-labelledby="hero-h" className="rounded-lg border border-border-subtle bg-surface p-6">
      <div className="mb-2 flex items-center gap-2">
        {/* D-H-05: pulsing red when count>0; honors prefers-reduced-motion via globals.css */}
        <span className="block h-2 w-2 rounded-full bg-severity-critical animate-pulse" aria-hidden="true" />
        <span className="text-xs uppercase tracking-wide text-text-muted">Action required</span>
      </div>
      <h2 id="hero-h" className="text-3xl font-semibold text-text">{headline}</h2>
      {subLine && (
        <p className="mt-2 line-clamp-2 text-base text-text-muted" title={subLine}>{subLine}</p>
      )}
      <div className="mt-4 flex flex-wrap gap-3">
        <Button asChild variant="cta" leftIcon={<Zap />}>
          <Link href="/dashboard/vulnerabilities?status=open&severity=critical">{microcopy.hero.ctaPrimary}</Link>
        </Button>
        {topVuln && (
          <Button variant="secondary" leftIcon={<Clock />} onClick={onSnooze} loading={snooze.isPending} loadingText="Snoozing…">
            {microcopy.hero.ctaSecondary}
          </Button>
        )}
      </div>
    </section>
  );
}
```

NOTE on the snooze id field: Plan 01 ships `top_vuln.id` (UUID) — the Hero uses `topVuln.id` directly. The matching backend route `POST /api/v1/vulnerabilities/{id}/unsnooze` (Plan 01 Task 3) drives the Undo path via `useUndoSnoozeMutation` (Plan 02). DO NOT fall back to cve_id — that would 404 on the backend (cve_id is not a primary key).

**`top5-card.tsx`** — D-T-01..05 + visual-language.md severity glyphs + SLA pill tiers:

```tsx
'use client';
import Link from 'next/link';
import { Card } from '@/components/ui/card';
import { useTopTriage } from '@/lib/queries/use-top-triage';
import { microcopy } from './microcopy';
import { cn } from '@/lib/utils';

const GLYPHS = { CRITICAL: '■', HIGH: '▲', MEDIUM: '◆', LOW: '○' } as const;
const GLYPH_COLOR = {
  CRITICAL: 'text-severity-critical',
  HIGH:     'text-severity-high',
  MEDIUM:   'text-severity-medium',
  LOW:      'text-severity-low',
} as const;

function slaPillClass(slaIso: string | null): string {
  if (!slaIso) return 'bg-surface-2 text-text-muted';
  const hours = (new Date(slaIso).getTime() - Date.now()) / 3_600_000;
  if (hours < 0)    return 'bg-danger-soft text-danger';        // breached
  if (hours < 72)   return 'bg-warning-soft text-warning';      // at risk
  return 'bg-success-soft text-success';                         // healthy
}

function fmtSla(slaIso: string | null): string {
  if (!slaIso) return '—';
  const hours = (new Date(slaIso).getTime() - Date.now()) / 3_600_000;
  if (hours < 0)  return `Breached ${Math.abs(Math.round(hours))}h ago`;
  if (hours < 48) return `${Math.round(hours)}h left`;
  return `${Math.round(hours / 24)}d left`;
}

export function Top5Card() {
  const q = useTopTriage(5);

  if (q.isPending) {
    return <Card><Card.Header><h2 className="text-lg font-semibold">{microcopy.top5.h2}</h2></Card.Header><div className="h-64 animate-pulse rounded-md bg-surface-2" aria-busy="true" /></Card>;
  }
  if (q.error) {
    const code = (q.error as any)?.code ?? 'unknown';
    const reqId = (q.error as any)?.requestId ?? 'unknown';
    return <Card><p role="alert" className="text-sm">{microcopy.error.inline('Top 5', code, reqId)}</p></Card>;
  }

  const items = q.data?.items ?? [];
  return (
    <Card>
      {/* D-Ax-01 — visible h2 'Top 5 to triage' (already correct; satisfies aria-labelledby contract). */}
      <Card.Header><h2 id="top5-h" className="text-lg font-semibold text-text">{microcopy.top5.h2}</h2></Card.Header>
      <ul aria-labelledby="top5-h" className="divide-y divide-border-subtle">
        {items.map((row) => (
          <li key={row.id}>
            <Link
              href={`/dashboard/vulnerabilities?cve=${encodeURIComponent(row.cve_id)}&open=drill`}
              className="grid grid-cols-[auto_1fr_auto_auto] items-center gap-3 px-1 py-3 hover:bg-surface-2 focus-visible:bg-surface-2 focus-visible:outline focus-visible:outline-2 focus-visible:outline-info"
            >
              <span className={cn('font-mono', GLYPH_COLOR[row.severity])} aria-label={row.severity.toLowerCase()}>{GLYPHS[row.severity]}</span>
              <div className="min-w-0">
                <p className="truncate font-mono text-sm text-text">{row.cve_id}</p>
                <p className="truncate font-mono text-xs text-text-muted">{row.host}</p>
              </div>
```
